# 根据组件分类 批量创建 md 说明文档，并写入标题

时间：2024年8月19日 10点44分

1. 根据导航栏 json 数据，使用 for 循环编写出 bat 脚本内容

```js
// 导航模拟数据
let nav = [
    {
        "text": "Advanced",
        "items": [
            { ...base_menu('向导', 'guide') },
            ...
        ]
    },
    ...
]

/*
 * 创建导航菜单的方法
 * @parms { title } 标题，即显示的内容
 * @parms { name } 名称 组件的英文，可带空格
 * @return {
 *     text: 菜单名称
 *     link: 路由
 * }
 * 这里的返参中 为了生成 bat内容 略有调整
 */

function base_menu(title, name) {
    const words = name.split(' ')
    const capitalizedWords = words.map(word =>
        word.charAt(0).toUpperCase() + word.slice(1).toLowerCase()
    )
    const linkString = capitalizedWords.join('_').toLowerCase()

    return {
        text: `${title}`,
        link: `${linkString}`
    }
}

let bat = `@echo off \n`
nav.forEach(item=>{
    bat += `mkdir ${item.text} \n`
    item.items.forEach(items=>{
        // echo # 向导 `guide` > Advanced\guide.md
        bat += `echo # ${items.text} \`${items.link}\` > ${item.text}\\${items.link}.md \n`
    })
})

console.log(bat)

```

将 `console.log` 中的 bat 复制到 bat 文件中，运行即可生成目录及文件
将目录和文件复制到对应文件夹即可

2. 修改 `.vitepress/config.js`

```js
// 重写路径
rewrites: {
    'components/:type/:name': 'components/:name',
}
```

即将原来路由 `components\Advanced\guide.md` 改为 `components\guide.md`
